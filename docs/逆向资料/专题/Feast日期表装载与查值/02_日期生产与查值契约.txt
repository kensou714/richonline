// C 是根对象 G+0C44h 的内嵌日期状态；63E590 返回这个子对象地址。

// 4052 写入边界
// ----------------------------------------------------------------------------
// 662DC0 先以包+2 WORD 调 607A2F -> 64F710；低 BYTE 为零就结束。
// 非零后取包+4 signed WORD 年、+6 signed BYTE 月、+7 signed BYTE 日。
// 以 ECX=G+0C44h 调 604CBC -> 694320。
// 694320 只窄写 WORD[C+8]、BYTE[C+10]、BYTE[C+11]，不修改 C+10h 查值槽。
// 已证读取下界 8 字节；这里没有消息长度、年份范围或月日合法性检查。
// GmsvID 校验的完整接受域沿用 4050 专题；不把包校验等同于日期合法性校验。
// getter：7D90E0 读 signed WORD[C+8]；727A70/727A90 读 signed BYTE[C+10/+11]。
//
// 7D8DE0 精确算法
// ----------------------------------------------------------------------------
// year_index = year - 2004;
// for (slot = 0; slot < 13; ++slot) {
//     offset = year_index * 26 + slot * 2;
//     if (sign_extend(table[offset]) == month &&
//         sign_extend(table[offset + 1]) == day) return slot;
// }
// return -1;
// 月日对两次使用 movsx byte；不是 unsigned BYTE 比较，也不是压缩日期整数。
// 不在本体检查 year_index，不排序，不去重，不按模式挑选日期，不返回所有命中槽。
// 返回契约：首个命中 0..12；无匹配 -1。月份或日期异常也没有独立拒绝分支。
//
// 当前容量与潜在错误索引
